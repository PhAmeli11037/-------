# 1.初始化仓库 git init # 2.绑定远程仓库（第一次关联） git remote add origin https://github.com/xxx/xxx.git # 3.查看状态 git status # 4.添加所有修改到暂存区 git add . # 5.提交代码，写备注 git commit -m "初次提交/功能更新" # 6.推送到远程 main 分支 git push origin main # 7.拉取远程最新代码 git pull origin main # 8.创建并切换新分支 git checkout -b dev # 9.切换已有分支 git checkout main
